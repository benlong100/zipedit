# ZipEdit v slovenščini.
#
# Machine-translated first, then corrected by Janez Starc over two passes
# (August and September 2026).
#
# Every word in here is now his. Of the three things the last round asked
# about, two came back answered -- he rewrote CHEATTXT outright and gave
# MSGWRAPPED an ordinary phrase -- and he moved the discard key besides.
#
# ONE IS STILL OPEN, and it is marked TODO where it sits:
#
#   WTAIL/WTAIL1 the word count, which no string in this file can make right
#
# Write real Slovenian here -- š, č, ž and all. The //e has no glyph for any
# of them, so tools/genlang.py transliterates on the way into the assembler
# and tools/xfer.sh turns them back into UTF-8 on the way to the Mac. You
# should never have to think about that; if you find yourself typing a { to
# mean an š, something has gone wrong and it is a bug in the tools.
#
# Rules, same as lang/en.txt: keep the names, mind the trailing spaces inside
# the quotes, and the two status rows are LAYOUTS whose columns must line up.

# ---------------------------------------------------------------- the document
# Becomes a real ProDOS filename if the writer saves without naming the file,
# so: fourteen characters at most -- the cell after it on the status row
# carries the unsaved-changes star -- a letter first, and no accented letters --
# ProDOS would not keep them and the disk is read by other programs.
UNTITLED = "BREZNASLOVA.MD"

# ------------------------------------------------------------- the cheat sheet
# Markdown syntax, which does not translate; only the words between the marks
# do. EXACTLY 80 characters at most -- the row is 80 wide and anything longer is
# simply cut off at the edge, mid-word.
#
# Rewritten by Janez on his second pass, after the first ran to 82 characters
# and lost "(url)" off the edge. His line is 70, so there are ten in hand.
CHEATTXT = "`koda` #H1 ##H2 naslovi 1. št. [ ] ali [x] opravilo > citat [link] url"

# --------------------------------------------------------------- the status row
# Exactly 80 characters, and the labels must stay at these columns because the
# numbers are written over the row at fixed positions:
#   col 38  the line label   col 45  the column label
#   col 52  free memory      col 64  the help hint
# The labels themselves are translated -- V: for vrstica, S: for stolpec -- but
# they must stay two characters and stay on those columns. A longer filename
# does not push them along; it eats the padding in front of them, and the
# digits land wherever geom.S says regardless.
STATTXT = " BREZNASLOVA.MD                       V:     S:     46K PROSTO  {OA}-? POMOČ       "

STATTXT40 = " BREZNASLOVA.MD  V:    S:               "

# Shown when a search ran off the end of the document and found its match by
# starting again at the top.
#
# Was our guess, " NADALJEVANO Z VRHA"; Janez replaced it on his second pass.
# The sense wanted is that the search reached the end and resumed at the
# beginning -- not that anything went wrong.
MSGWRAPPED = " NADALJUJEM IZ VRHA"

MSGCOPIED = " VRSTICA KOPIRANA"

# TODO -- the word count is wrong in Slovenian and the strings cannot fix it.
# Slovenian counts in four forms: 1 beseda, 2 besedi, 3-4 besede, 5+ besed.
# The editor has only two, singular and plural, so "2 BESED" and "3 BESED"
# will both be wrong. Fixing it means changing src/edit_ops.S, not this file.
# " BESED" is the form used with 5 and above, which is the commonest case.
WTAIL1 = " BESEDA"
WTAIL = " BESED"

# ------------------------------------------------------------------- the guard
# The key letters ARE translated, and they are two lines below. This used to
# say the opposite: S and D were fixed in the source, and the Slovenian had to
# be chosen to fit round them. Janez pointed out that D carries nothing in a
# language where the word is "zavrzi", so the letters moved into this file and
# the program now reads them from here.
#
# Whatever these two lines name, KEYSAVE and KEYDISCARD must match.
MSGUNSAVED = " NESHRANJENE SPREMEMBE.  S = SHRANI   Z = ZAVRZI   ESC = PREKLIČI"
MSGUNSV40 = " Shranim?  S shrani  Z zavrzi  Esc"

# Kateri tipki ponuja vprašanje zgoraj -- ena velika črka na tipko, in morata
# biti tisti, ki ju vrstici zgoraj imenujeta. Janez je zavrzi premaknil z D na
# Z, ker se beseda začne z Z; to je koda, ki mu zdaj sledi.
KEYSAVE = "S"
KEYDISCARD = "Z"
